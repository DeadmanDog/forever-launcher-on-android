cmake_minimum_required(VERSION 3.22.1)
project(FrontMageVulkanProbe LANGUAGES C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CPP "${REPO}/app/src/main/cpp")
add_subdirectory("${CPP}/extras/adrenotools" adrenotools EXCLUDE_FROM_ALL)
add_library(wow_vkd3d_probe SHARED "${CPP}/extras/WowVkd3dProbe.cpp")
target_compile_features(wow_vkd3d_probe PRIVATE cxx_std_17)
target_compile_options(wow_vkd3d_probe PRIVATE -O2 -fvisibility=hidden)
target_link_libraries(wow_vkd3d_probe adrenotools dl log android)
target_link_options(wow_vkd3d_probe PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)

cmake_minimum_required(VERSION 3.22.1)
project(WowVulkanRenderer LANGUAGES C CXX)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CPP "${REPO}/app/src/main/cpp")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CPP}/extras/adrenotools" adrenotools EXCLUDE_FROM_ALL)
find_program(GLSLANG glslangValidator REQUIRED)
foreach(shader window_vert window_frag)
  if(shader STREQUAL "window_vert")
    set(suffix vert)
  else()
    set(suffix frag)
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${shader}.h"
    COMMAND ${GLSLANG} -V --vn ${shader}_code -x -o "${CMAKE_CURRENT_BINARY_DIR}/${shader}.h" "${CPP}/winlator/window.${suffix}"
    DEPENDS "${CPP}/winlator/window.${suffix}")
endforeach()
set(RENDERER_FILES vulkan_jni.cpp VulkanRendererContext.cpp VulkanRendererScanout.cpp)
foreach(f IN LISTS RENDERER_FILES)
  list(APPEND SOURCES "${CPP}/winlator/${f}")
endforeach()
add_library(vulkan_renderer SHARED ${SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/window_vert.h" "${CMAKE_CURRENT_BINARY_DIR}/window_frag.h")
target_include_directories(vulkan_renderer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CPP}/winlator")
set_target_properties(vulkan_renderer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(vulkan_renderer PRIVATE -O2 -fvisibility=hidden)
target_link_libraries(vulkan_renderer log android vulkan adrenotools dl atomic)
target_link_options(vulkan_renderer PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
